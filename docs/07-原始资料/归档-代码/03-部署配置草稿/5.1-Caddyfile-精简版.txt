# 网站 A：莫伯资本 (MoBoCapital.com)
mobocapital.com, www.mobocapital.com {
    root * /srv/sites/mobocapital
    encode zstd gzip

    @internal_zone {
        path /internal/*
    }
    handle @internal_zone {
        basicauth {
            mobo {$CADDY_INTERNAL_PASS_HASH}
        }
        header Cache-Control "no-store, no-cache, must-revalidate"
        file_server
    }

    @public_json {
        path /data/*.json
    }
    handle @public_json {
        header Cache-Control "public, max-age=60, must-revalidate"
        file_server
    }

    handle {
        try_files {path} /index.html
        file_server
    }

    header {
        Strict-Transport-Security "max-age=31536000; includeSubDomains; preload"
        X-Content-Type-Options "nosniff"
        X-Frame-Options "SAMEORIGIN"
        Referrer-Policy "strict-origin-when-cross-origin"
    }
}

# 网站 B：莫伯之家 (MoBoHome.com)
mobohome.com, www.mobohome.com {
    root * /srv/sites/mobohome
    encode zstd gzip

    @home_internal {
        path /internal/*
    }
    handle @home_internal {
        basicauth {
            mobo {$CADDY_INTERNAL_PASS_HASH}
        }
        header Cache-Control "no-store, no-cache, must-revalidate"
        file_server
    }

    handle_path /lab/* {
        basicauth {
            mobo {$CADDY_INTERNAL_PASS_HASH}
        }
        reverse_proxy archive-dashboard:8501 {
            header_up Host {host}
            header_up X-Real-IP {remote_host}
        }
    }

    handle {
        try_files {path} /index.html
        file_server
    }

    header {
        Strict-Transport-Security "max-age=31536000; includeSubDomains; preload"
        X-Content-Type-Options "nosniff"
        X-Frame-Options "SAMEORIGIN"
    }
}
